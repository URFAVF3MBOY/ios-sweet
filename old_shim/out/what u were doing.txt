static unsigned int
osm_cpufreq_fast_switch(struct cpufreq_policy *policy, unsigned int target_freq)
{
	int index;

	index = cpufreq_frequency_table_target(policy, target_freq,
							CPUFREQ_RELATION_L);
	if (index < 0)
		return 0;

	osm_cpufreq_target_index(policy, index);

	return policy->freq_table[index].frequency;
}

static unsigned int osm_cpufreq_get(unsigned int cpu)
{

	if (!policy)

}

/*
 * OP7T stage 0 (v2): read-only hardware probe - CPU speed, CPU DVFS (OSM) tables and state,
 * Adreno power state.
 *
 * Install as bsd/dev/sweet_hwprobe.c.  It never writes a register.  Blocks that could hang
 * the bus when gated (the GPU clock controller) are read only after a status register in an
 * always-on block says they are powered and clocked.
 *
 * Output: "SWEET HW:" lines on the console / live log / persistent log.
 * Boot-arg: sweethwprobe=0 disables it (default on).
 *
 * Register layout VERIFIED against the OnePlus SM8150 kernel source you supplied:
 *   clk-cpu-osm.c : ENABLE 0x0, CORE_DCVS_CTRL 0xbc, FREQ_REG 0x110 + i*32 (40 entries),
 *                   VOLT_REG 0x114 + i*32, DCVS_PERF_STATE_DESIRED_REG(n) 0x920 + 4n,
 *                   OSM_CYCLE_COUNTER_STATUS_REG(n) 0x9c0 + 4n.
 *                   FREQ_REG: src = bits 31:30 (0 => 300 MHz), lval = bits 7:0 (x 19.2 MHz),
 *                   core count = bits 18:16.  VOLT_REG: corner = bits 21:16, volt = bits 11:0.
 *   gcc-sm8150.c  : GPU_CFG_AHB 0x71004, GPU_MEMNOC_GFX 0x7100c, GPU_SNOC_DVM_GFX 0x71018.
 *   device tree   : gpu_cx_gdsc 0x02c9106c, gpu_gx_gdsc 0x02c9100c.
 */
#if defined(ARM64_BOARD_CONFIG_SWEET)
#include <sys/param.h>
#include <sys/systm.h>
#include <pexpert/pexpert.h>

extern vm_offset_t ml_io_map(vm_offset_t, vm_size_t);
void sweet_hwprobe(void);

#define GCC_PA              0x00100000
#define GCC_SIZE            0x00090000
#define GCC_GPU_CFG_AHB     0x71004
#define GCC_GPU_MEMNOC_GFX  0x7100c
#define GCC_GPU_SNOC_DVM    0x71018
#define CBCR_CLK_OFF        (1u << 31)

#define GPUCC_PA            0x02c90000
#define GPUCC_SIZE          0x00009000
#define GPUCC_CX_GDSCR      0x106c
#define GPUCC_GX_GDSCR      0x100c

#define OSM_SIZE            0x1400
#define OSM_ENABLE          0x000
#define OSM_DCVS_CTRL       0x0bc
#define OSM_FREQ(i)         (0x110 + (i) * 32)
#define OSM_VOLT(i)         (0x114 + (i) * 32)
#define OSM_DESIRED(n)      (0x920 + 4 * (n))
#define OSM_CYCLES(n)       (0x9c0 + 4 * (n))
#define OSM_TABLE_SIZE      40

static const struct {
	const char *name;
	uint32_t    pa;
	unsigned    ncores;     /* cores that have per-core registers in this block */
	const char *cpus;
} osm_blocks[] = {
	{ "l3",    0x18321000, 1, "L3" },
	{ "pwr",   0x18323000, 4, "cpu0-3" },
	{ "perf",  0x18325800, 3, "cpu4-6" },
	{ "prime", 0x18327800, 1, "cpu7" },
};
#define NBLOCKS (sizeof(osm_blocks) / sizeof(osm_blocks[0]))

static vm_offset_t osm_base[NBLOCKS];

static uint32_t
hw_rd(vm_offset_t base, uint32_t off)
{
	uint32_t v = *(volatile uint32_t *)(base + off);
	__asm__ volatile ("dmb oshld" ::: "memory");
	return v;
}

/* Frequency of one LUT word in kHz (same decode as the OnePlus driver). */
static uint32_t
osm_khz(uint32_t w)
{
	uint32_t src = (w >> 30) & 3, lval = w & 0xff;
	return src ? 19200 * lval : 300000;
}

/*
 * Estimate the boot CPU's clock: a chain of dependent adds retires one add per cycle on both
 * the A55 and the A76, so cycles = 16 * iterations.  Also samples cpu0's OSM cycle counter
 * (pwr block, core 0) around the loop as a cross-check.
 */
static void
hw_cpu_mhz(void)
{
	uint64_t frq, t0, t1, iters = 12500000ULL, x = 0, n = iters;
	uint32_t c0 = 0, c1 = 0;
	vm_offset_t pwr = osm_base[1];

	__asm__ volatile ("mrs %0, cntfrq_el0" : "=r"(frq));
	if (pwr) {
		c0 = hw_rd(pwr, OSM_CYCLES(0));
	}
	__asm__ volatile ("isb\n\tmrs %0, cntvct_el0" : "=r"(t0) :: "memory");
	__asm__ volatile (
		"1:\n\t"
		"add %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\t"
		"add %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\t"
		"add %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\t"
		"add %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\tadd %0, %0, #1\n\t"
		"subs %1, %1, #1\n\tb.ne 1b"
		: "+r"(x), "+r"(n) :: "cc");
	__asm__ volatile ("isb\n\tmrs %0, cntvct_el0" : "=r"(t1) :: "memory");
	if (pwr) {
		c1 = hw_rd(pwr, OSM_CYCLES(0));
	}
	if (t1 > t0 && frq != 0) {
		uint64_t ticks = t1 - t0;
		uint64_t mhz_x10 = ((16ULL * iters * 10ULL) / ticks) * frq / 1000000ULL;
		printf("SWEET HW: cpu0 dependent-add loop: %llu ticks at %llu Hz timer, ~%llu.%llu MHz, x=%llu\n",
		    ticks, frq, mhz_x10 / 10, mhz_x10 % 10, x);
		if (pwr) {
			uint32_t dc = c1 - c0;
			uint64_t cmhz_x10 = ((uint64_t)dc * 10ULL / ticks) * frq / 1000000ULL;
			printf("SWEET HW: cpu0 OSM cycle counter %u -> %u (delta %u), ~%llu.%llu MHz\n",
			    c0, c1, dc, cmhz_x10 / 10, cmhz_x10 % 10);
		}
	} else {
		printf("SWEET HW: cpu0 speed estimate failed (t0 %llu t1 %llu frq %llu)\n", t0, t1, frq);
	}
}

static void
hw_osm(void)
{
	for (unsigned b = 0; b < NBLOCKS; b++) {
		vm_offset_t base = osm_base[b];
		uint32_t lut_w[OSM_TABLE_SIZE];
		unsigned used = 0;

		if (base == 0) {
			printf("SWEET HW: OSM %s map failed\n", osm_blocks[b].name);
			continue;
		}
		printf("SWEET HW: OSM %s (%s) @0x%08x enable 0x%08x dcvs_ctrl 0x%08x (per-core %s)\n",
		    osm_blocks[b].name, osm_blocks[b].cpus, osm_blocks[b].pa,
		    hw_rd(base, OSM_ENABLE), hw_rd(base, OSM_DCVS_CTRL),
		    (hw_rd(base, OSM_DCVS_CTRL) & 1) ? "yes" : "no");
		for (unsigned i = 0; i < OSM_TABLE_SIZE; i++) {
			uint32_t w = hw_rd(base, OSM_FREQ(i));
			uint32_t v = hw_rd(base, OSM_VOLT(i));
			lut_w[i] = w;
			used = i + 1;
			printf("SWEET HW: OSM %s lut[%2u] freq %7u kHz  raw 0x%08x  corner %2u volt %4u\n",
			    osm_blocks[b].name, i, osm_khz(w), w, (v >> 16) & 0x3f, v & 0xfff);
			if (i > 0 && osm_khz(w) == osm_khz(lut_w[i - 1])) {
				break;          /* two equal entries mark the end of the table */
			}
		}
		for (unsigned n = 0; n < osm_blocks[b].ncores; n++) {
			uint32_t idx = hw_rd(base, OSM_DESIRED(n));
			uint32_t khz = (idx < used) ? osm_khz(lut_w[idx]) : 0;
			printf("SWEET HW: OSM %s core %u desired index %u -> %u kHz, cycle counter 0x%08x\n",
			    osm_blocks[b].name, n, idx, khz, hw_rd(base, OSM_CYCLES(n)));
		}
	}
}

static void
hw_gpu(void)
{
	vm_offset_t gcc = ml_io_map(GCC_PA, GCC_SIZE);
	if (gcc == 0) {
		printf("SWEET HW: GCC map failed\n");
		return;
	}
	uint32_t cfg = hw_rd(gcc, GCC_GPU_CFG_AHB);
	uint32_t mem = hw_rd(gcc, GCC_GPU_MEMNOC_GFX);
	uint32_t dvm = hw_rd(gcc, GCC_GPU_SNOC_DVM);
	printf("SWEET HW: GCC gpu_cfg_ahb 0x%08x (%s) memnoc_gfx 0x%08x (%s) snoc_dvm 0x%08x (%s)\n",
	    cfg, (cfg & CBCR_CLK_OFF) ? "off" : "on",
	    mem, (mem & CBCR_CLK_OFF) ? "off" : "on",
	    dvm, (dvm & CBCR_CLK_OFF) ? "off" : "on");
	if (cfg & CBCR_CLK_OFF) {
		printf("SWEET HW: gpu_cfg_ahb is off; GPUCC not read (a gated read would hang)\n");
		return;
	}
	vm_offset_t cc = ml_io_map(GPUCC_PA, GPUCC_SIZE);
	if (cc == 0) {
		printf("SWEET HW: GPUCC map failed\n");
		return;
	}
	uint32_t cx = hw_rd(cc, GPUCC_CX_GDSCR);
	uint32_t gx = hw_rd(cc, GPUCC_GX_GDSCR);
	printf("SWEET HW: GPUCC CX_GDSCR 0x%08x (%s) GX_GDSCR 0x%08x (%s)\n",
	    cx, (cx & (1u << 31)) ? "on" : "off", gx, (gx & (1u << 31)) ? "on" : "off");
	/* The Adreno register block (0x02c00000) is deliberately not read: it needs CX and GX on. */
}

void
sweet_hwprobe(void)
{
	int enabled = 1;

	PE_parse_boot_argn("sweethwprobe", &enabled, sizeof(enabled));
	if (!enabled) {
		return;
	}
	printf("SWEET HW: stage 0 probe v2 start (read-only)\n");
	for (unsigned b = 0; b < NBLOCKS; b++) {
		osm_base[b] = ml_io_map(osm_blocks[b].pa, OSM_SIZE);
	}
	hw_cpu_mhz();
	hw_osm();
	hw_gpu();
	printf("SWEET HW: stage 0 probe done\n");
}
#endif /* ARM64_BOARD_CONFIG_SWEET */
